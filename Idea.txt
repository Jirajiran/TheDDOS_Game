Data-Driven Design (การออกแบบที่ขับเคลื่อนด้วยข้อมูล) — อธิบายฝั่งโครงสร้างโค้ดและ Editor
คือการแยก "ลอจิกของโค้ด" ออกจาก "ข้อมูล (Data / Parameters)" โดยไม่เขียน Hardcode ตายตัวลงไปในโค้ดว่าเกมนี้มีคลาส Tank หรือ Runner แต่สร้างเทมเพลตกลางที่มีตัวแปร (HP, Speed) เปิดรับค่า Input ผ่านหน้า Editor (เช่น Inspector ใน Unity/Unreal หรือไฟล์ JSON) เพื่อให้ Game Designer สามารถปรับเปลี่ยนตัวเลขและสร้างยูนิตใหม่ๆ ได้ไม่จำกัดโดยไม่ต้องให้โปรแกรมเมอร์แก้โค้ด

Emergent Design / Emergent Gameplay (การออกแบบเชิงอุบัติการณ์) — อธิบายฝั่งการเกิดบทบาทในเกม
คือปรากฏการณ์ที่ระบบมีเพียงกฎหรือค่าตัวแปรพื้นฐานง่ายๆ แต่เมื่อนำค่าเหล่านั้นมาผสมผสานกันในสัดส่วนที่ต่างกัน กลับก่อให้เกิด "บทบาท (Archetype)" หรือ "พฤติกรรมการเล่น" ใหม่ๆ ขึ้นมาเองตามธรรมชาติในสายตาของผู้เล่น ทั้งที่ในระบบโค้ดไม่เคยมีการเขียนนิยามคำว่า Tank หรือ นักวิ่ง เอาไว้เลย

Composition over Inheritance / ECS (Entity Component System) — แนวคิดสถาปัตยกรรมโค้ดที่รองรับ
ในเชิงวิศวกรรมซอฟต์แวร์เกม การเลี่ยงไม่สร้างคลาสลูกแยกประเภท (class Tank : Unit, class Runner : Unit) แต่ใช้การประกอบคุณสมบัติและตัวแปรเข้าไปใน Entity เดียวกัน ถือเป็นหลักการของ Composition และ ECS ซึ่งเป็นโครงสร้างพื้นฐานที่ทำให้ระบบแบบ Data-Driven ทำงานได้อย่างยืดหยุ่นสูงสุด

หากเรียกให้กระชับที่สุดในวงการพัฒนาเกม การวางระบบแบบนี้คือการใช้ Data-Driven Design เพื่อเปิดทางให้เกิด Emergent Archetypes (บทบาทที่อุบัติขึ้นเองจากค่าพารามิเตอร์)

ตัวอย่างโค้ดเปรียบเทียบให้เห็นภาพชัดเจนระหว่างการเขียนเกมแบบ Hardcode กับแบบ Data-Driven ที่เปิดรับค่าผ่าน Editor
เปรียบเทียบด้วยภาษา C# (สไตล์ Unity Engine) โดยจำลองสถานการณ์สร้างยูนิตที่มีตัวแปร hp และ walkSpeed ตามที่คุณยกตัวอย่างครับ1. แบบ Hardcode (ฝังค่าและบทบาทตายตัวในโค้ด)ในแบบนี้ โปรแกรมเมอร์ต้องเขียนดักเงื่อนไขเอาไว้ล่วงหน้าว่าเกมมีตัวละครประเภทไหนบ้าง และแต่ละประเภทมีค่าพลังเท่าไหร่C#using UnityEngine;

public class UnitHardcode : MonoBehaviour
{
    public string roleName; // ต้องระบุชื่อประเภทให้ตรงกับในโค้ด
    private int hp;
    private float walkSpeed;

    void Start()
    {
        // ❌ Hardcode: ฝังทั้ง "ชื่อบทบาท" และ "ตัวเลข" ลงไปในลอจิก
        if (roleName == "Tank")
        {
            hp = 1000;
            walkSpeed = 2.0f;
        }
        else if (roleName == "Runner")
        {
            hp = 100;
            walkSpeed = 10.0f;
        }
        // ถ้าอยากเพิ่มตัวละคร "นักธนู" หรือ "บอส" ต้องกลับมาแก้ไฟล์โค้ดนี้เสมอ
    }

    void Update()
    {
        transform.Translate(Vector3.forward * walkSpeed * Time.deltaTime);
    }
}
จุดสังเกต: โค้ดรู้จักคำว่า "Tank" และ "Runner" ถ้า Game Designer อยากลองปรับเลือด Tank จาก 1000 เป็น 1200 หรืออยากสร้างยูนิตใหม่ที่วิ่งเร็วปานกลาง เลือดปานกลาง จะทำเองไม่ได้ ต้องรอโปรแกรมเมอร์แก้โค้ดและ Compile ใหม่เท่านั้น2. แบบ Data-Driven (เปิดรับค่าผ่าน Editor ไม่มีนิยามบทบาทในโค้ด)ในแบบนี้ โค้ดจะทำหน้าที่เป็นเพียง "โครงสร้างระบบกลาง" ที่ไม่สนใจว่ายูนิตตัวนี้คืออาชีพอะไร สนใจแค่การนำตัวเลขที่ได้รับจาก Editor ไปคำนวณC#using UnityEngine;

public class UnitDataDriven : MonoBehaviour
{
    // ✅ Data-Driven: ใช้ [SerializeField] เปิดช่องให้กรอกค่าผ่านหน้าต่าง Inspector (Editor)
    [SerializeField] private int hp = 100;
    [SerializeField] private float walkSpeed = 5.0f;

    // สังเกตว่าในโค้ดทั้งหมด "ไม่มี" คำว่า Tank หรือ Runner อยู่เลย
    void Update()
    {
        // โค้ดทำงานตามตัวเลขที่ถูกป้อนเข้ามาอย่างเดียว
        transform.Translate(Vector3.forward * walkSpeed * Time.deltaTime);
    }

    public void TakeDamage(int damage)
    {
        hp -= damage;
        if (hp <= 0)
        {
            Destroy(gameObject);
        }
    }
}
สิ่งที่เกิดขึ้นจริงในหน้า Editor (ไม่ต้องแตะโค้ดอีกเลย):จากโค้ด UnitDataDriven เพียงไฟล์เดียว ทีมออกแบบสามารถกดสร้าง Prefab (แม่แบบวัตถุใน Editor) แยกออกมาได้ไม่จำกัด:สร้าง Prefab 1: ลากสคริปต์นี้ใส่ แล้วพิมพ์ในช่อง Editor ว่า Hp = 1000, Walk Speed = 2 $\rightarrow$ ผู้เล่นเห็นแล้วเรียกว่า Tankสร้าง Prefab 2: ลากสคริปต์เดิมใส่ แล้วพิมพ์ในช่อง Editor ว่า Hp = 100, Walk Speed = 10 $\rightarrow$ ผู้เล่นเห็นแล้วเรียกว่า นักวิ่งสร้าง Prefab 3: ลากสคริปต์เดิมใส่ แล้วพิมพ์ในช่อง Editor ว่า Hp = 5000, Walk Speed = 8 $\rightarrow$ ผู้เล่นเห็นแล้วเรียกว่า บอสจุดแตกต่างสำคัญที่เห็นได้จากโค้ดทั้ง 2 แบบการตัด if-else ที่เช็กประเภททิ้ง: โค้ดแบบ Data-Driven จะไม่มี if (type == ...) เพื่อแยกสายอาชีพ แต่ใช้ตัวแปรตัวเดียวกันรองรับทุกความเป็นไปได้การแยกหน้าที่ (Separation of Concerns): โค้ดมีหน้าที่แค่ "ทำให้วัตถุเคลื่อนที่ตามความเร็วและตายเมื่อเลือดหมด" ส่วนหน้าที่ "นิยามว่าตัวไหนคือ Tank หรือนักวิ่ง" ถูกโยนไปให้ข้อมูล (Data) ในหน้า Editor เป็นตัวกำหนดหากต้องการขยับไปอีกขั้นในโปรเจกต์ขนาดใหญ่ นักพัฒนามักจะแยกข้อมูล hp และ walkSpeed ออกจากตัวละครไปเก็บไว้ใน ScriptableObject หรือไฟล์ตาราง JSON / CSV เพื่อให้สามารถแก้ไขค่าตัวละครเป็นร้อยตัวพร้อมกันได้ในไฟล์เดียวโดยไม่ต้องไล่คลิกแก้ทีละ Prefab ครับ
